cmake_minimum_required(VERSION 3.20)
project(mini_exchange LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(MX_BUILD_TESTS "Build tests" ON)
option(MX_BUILD_BENCH "Build benchmarks" ON)
option(MX_SANITIZE "Enable ASan and UBSan" OFF)
option(MX_TSAN "Enable TSan" OFF)
option(MX_NATIVE "Compile with native CPU tuning" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion)
if(MX_NATIVE)
  add_compile_options(-march=native)
endif()
if(MX_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(MX_TSAN)
  add_compile_options(-fsanitize=thread -g)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
add_library(mx_core INTERFACE)
target_include_directories(mx_core INTERFACE src)
target_link_libraries(mx_core INTERFACE Threads::Threads)
